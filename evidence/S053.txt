def envelope_spectrum(x, fs, band=(2000, 5000), fmax=300.0):

def coral_loss(Xs, Xt): 
    
if Xs.shape[0] < 2 or Xt.shape[0] < 2: 
        return 0.0 
    Xs = Xs - Xs.mean(0, keepdims=True) 
    Xt = Xt - Xt.mean(0, keepdims=True) 
    Cs = np.cov(Xs, rowvar=False) 
    Ct = np.cov(Xt, rowvar=False) 
    if (not np.isfinite(Cs).all()) or (not np.isfinite(Ct).all()): 
        return 0.0 
    return float(np.linalg.norm(Cs - Ct, ord='fro')**2)
